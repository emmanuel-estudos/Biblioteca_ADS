def filtro_8bit(img, fator=8, bits=2):
    """Simula gráficos de consoles 8-bit com pixels grandes e poucas cores.
    """
    # pegando altura e largura originais
    h, w = img.shape[:2]

    # reduzindo resolução
    h_reduzida = h // fator
    w_reduzida = w // fator
    img_reduzida = cv2.resize(img, (w_reduzida, h_reduzida), interpolation=cv2.INTER_AREA)

    # amplicar para o tamanho original
    img_pixelada = cv2.resize(img_reduzida, (w, h), interpolation=cv2.INTER_NEAREST)

    # quantização de cores dos 3 canais
    niveis = 2 ** bits # número de níveis por canal, de acordo com o número de bits
    tamanho_intervalo = 256 / niveis

    # aplicando a quantificação de forma vetorizada para todos os canais
      ## dividindo pelo intervalo, convertendo para inteiro, reescalonando para a faixa de 0-255
    img_quantizada = (img_pixelada // tamanho_intervalo) * (255 / (niveis - 1))

    # convertendo novamente para uint8
    saida = img_quantizada.astype(np.uint8)

    return saida

    # raise NotImplementedError

- **h, w = img.shape[:2]**: pegando a altura e a largura originais da foto
- **h_reduzida = h // fator** e **w_reduzida = w // fator**: dividem as dimensões pelo `fator` definido na chamada da função.
- **img_reduzida = cv2.resize(img, (w_reduzida, h_reduzida), interpolation=cv2.INTER_AREA)**: faz uma interpolação e diminui a imagem por meio das médias dos pixeis da região
- **img_pixelada = cv2.resize(img_reduzida, (w, h), interpolation=cv2.INTER_NEAREST)**: a imagem antes reduzida é esticada para o tamanho original
- **niveis = 2 ** bits**: Com `bits=2`, temos $2^2 = 4$ tons possíveis por canal de cor (Azul, Verde e Vermelho separadamente)
- **tamanho_intervalo = 256 / niveis**: Divide o espectro total de 256 tons (0 a 255) em blocos (com bits=2, cada bloco tem tamanho $256 / 4 = 64$)
- img_quantizada = (img_pixelada // tamanho_intervalo) * (255 / (niveis - 1))
  - **img_pixelada // tamanho_intervalo** agrupa os valores de cor contínuos em faixas discretas (ex: 0, 1, 2, 3)
  - **(255 / (niveis - 1))** reescala esses valores para que voltem a ocupar o intervalo total de intensidade de 0 a 255, mas agora limitados a apenas poucos degraus de cor bem definidos

def caixa_de_dialogo(img, texto="quando me levanto muito rapido, vejo assim"):
    """Cria uma tela de RPG aplicando o filtro 8-bit e desenhando uma caixa de
    diálogo com borda e texto na parte inferior.

def zoom_dramatico(img, centro=None, niveis=3, tamanho=300):
  """Cria um painel lado a lado com recortes progressivamente menores
    (zoom dramático) centrados em um ponto da imagem.
  """

  # pegando as dimensões da imagem
  h, w = img.shape[:2]

  # caso o centro não foi fornecido, pega o centro da imagem
  if centro is None:
      cx, cy = w // 2, h // 2
  else:
      cx, cy = centro

  # Definir o tamanho do recorte inicial
    ## (ex: metade da menor dimensão da imagem)
  tamanho_recorte_inicial = min(h, w) // 2

  # criando array onde ficarão as imagens
  quadros = []

  # 'for' para ir reduzindo recursivamente
  for i in range(niveis):
        # a cada nível, o tamanho do recorte cai pela metade (o zoom dobra)
        l = int(tamanho_recorte_inicial / (2 ** i))
        l = max(10, l)  # Garante um tamanho mínimo seguro

        # Calcular coordenadas do retângulo centrado em (cx, cy)
        x1 = max(0, cx - l // 2)
        y1 = max(0, cy - l // 2)
        x2 = min(w, cx + l // 2)
        y2 = min(h, cy + l // 2)

        # extrai a Região de Interesse (ROI)
        roi = img[y1:y2, x1:x2]

        # redimensiona o recorte para o tamanho padrão do painel
        quadro_redimensionado = cv2.resize(roi, (tamanho, tamanho), interpolation=cv2.INTER_LINEAR)

        quadros.append(quadro_redimensionado)

  # Juntar todos os quadros lado a lado horizontalmente
  painel = np.hstack(quadros)

  return painel

  # raise NotImplementedError

_ = mostrar_resultado(zoom_dramatico, img_2_2)
```

- **h, w = img.shape[:2]**: pegando as dimensões originais
- **if centro is None:**: calcula o centro caso ele não seja fornecido
- **tamanho_recorte_inicial = min(h, w) // 2**: define onde vai ser o primeiro recorte (metade da menor dimensão da imagem)
- **quadros = []**: cria um array para guardar as imagens

- **painel = np.hstack(quadros)**: junta todos os quadros

def recorte_quadrado(img, lado=512):
    """Recorta o maior quadrado central de uma imagem e
    a redimensiona para lado x lado."""

    # pegando o tamanho da imagem
    h, w = img.shape[:2]

    # verificando o tamanho do menor lado para garantir um quadrado perfeito central
    tamanho_min = min(h, w)

    # calculando as coordenadas centrais para o corte
    y_inicio = (h - tamanho_min) // 2
    x_inicio = (w - tamanho_min) // 2

    # recortando a região quadrada central
    recorte = img[y_inicio:y_inicio + tamanho_min, x_inicio:x_inicio + tamanho_min]

    # redicionamento para o tamanho final desejado
    img_quadrada = cv2.resize(recorte, (lado, lado), interpolation=cv2.INTER_AREA)

    return img_quadrada
    # raise NotImplementedError


def figurinha(img, texto="estou assim", lado=512):
    """Cria uma figurinha BGRA com recorte circular suave (canal alfa)
    e uma legenda inferior com contorno."""

    # deixa a iamgem quadrada, usando a função auxiliar
    img_quadrada = recorte_quadrado(img, lado)

    # convertendo BGR para BGRA (canal de transparência)
    img_bgra = cv2.cvtColor(img_quadrada, cv2.COLOR_BGR2BGRA)

    # criando círculo
    y, x = np.mgrid[:lado, :lado]
    centro = lado / 2.0
    raio = lado / 2.0 - 10

    # calculando a distância de cada pixel para o centro
    distancia = np.sqrt((x - centro) ** 2 + (y - centro) ** 2)

    # criando suavização na borda
    alfa = np.clip((raio - distancia) + 1.5, 0, 1.5) / 1.5
    alfa = (alfa * 255).astype(np.uint8)

    # aplicando 'alfa' na imagem BGRA
    img_bgra[:, :, 3] = alfa

    # adicionando a legenda
    texto = texto.upper()
    fonte = cv2.FONT_HERSHEY_DUPLEX
    escala = lado / 600
    esp_contorno = max(2, int(escala * 4))
    esp_texto = max(1, int(escala * 2))

    # garantindo que o texto caiba na figurinha (largura)
    while cv2.getTextSize(texto, fonte, escala, esp_contorno)[0][0] > lado - 30:
        escala -= 0.05
        if escala <= 0.2:
            break

    (tw, th), _ = cv2.getTextSize(texto, fonte, escala, esp_contorno)
    tx = (lado - tw) // 2
    ty = lado - int(lado * 0.1)

    # cores na legenda
      ## também com 4 canais
    cor_preta = (0, 0, 0, 255)
    cor_branca = (255, 255, 255, 255)

    # colocando texto na imagem
    cv2.putText(img_bgra, texto, (tx, ty), fonte, escala, cor_preta, esp_contorno, cv2.LINE_AA)
    cv2.putText(img_bgra, texto, (tx, ty), fonte, escala, cor_branca, esp_texto, cv2.LINE_AA)

    return img_bgra

def faixa_de_foco(img, centro=0.6, largura=0.15, transicao=0.25):
    """Gera uma máscara 2D de foco com uma faixa horizontal nítida
    e transição gradual (efeito Tilt-Shift / miniatura)."""

    # pegando as dimensões da imagem
    h, w = img.shape[:2]

    # colocando frações em coordenadas
    y_centro = centro * h
    meia_largura = (largura * h) / 2.0
    t_transicao = max(1.0, transicao * h)  # evita divisão por zero

    # criando um vetor de coordenadas y para cada linha da imagem (shape: (h,))
    y = np.arange(h, dtype=np.float32)

    # calculando a distância absoluta de cada linha em relação aos limites da faixa de foco
      ## Distância é 0 se estiver dentro da faixa nítida, e positiva se estiver fora
    distancia = np.maximum(0, np.abs(y - y_centro) - meia_largura)

    # calculando a opacidade/máscara com base na transição gradual
      ## distância normalizada de 0 (foco total) a 1 (fora da transição)
    mascara_linha = 1.0 - np.clip(distancia / t_transicao, 0.0, 1.0)

    # expandindo para 2D usando broadcasting (replicando o valor da linha para todas as colunas w)
      ## resultado shape: (h, w) com valores de 0.0 a 1.0
    mascara_2d = np.tile(mascara_linha[:, np.newaxis], (1, w))

    # convertendo para uint8 na faixa 0 a 255
    mascara_uint8 = (mascara_2d * 255).astype(np.uint8)

    return mascara_uint8

_ = mostrar_resultado(faixa_de_foco, img_3_1)
```

- **h, w = img.shape[:2]**: pegando as dimensões originais da imagem
- **y_centro = centro * h**, **meia_largura = (largura * h) / 2.0** e **t_transicao = max(1.0, transicao * h)**: colocam os parâmetros passados em coordenadas referentes a imagem
- **y = np.arange(h, dtype=np.float32)**: vetor para as coordenadas verticais das linhas da imagem
- **distancia = np.maximum(0, np.abs(y - y_centro) - meia_largura)**: calcula a distância de cada linha para o `limite` da faixa de foco (0 = dentro e positiva = distância)
- **mascara_linha = 1.0 - np.clip(distancia / t_transicao, 0.0, 1.0)**: cria a máscara de acordo com a `transição` (0 = no foco, positivo = distância do foco e 1 = fora)
- **mascara_2d = np.tile(mascara_linha[:, np.newaxis], (1, w))**: transformando em 2D para a criação da imagem
- **mascara_uint8 = (mascara_2d * 255).astype(np.uint8)**: converte a matriz criada no passo anterior para `unit8`.

def desfoque_reamostragem(img, fator=6, passadas=2):
    """Gera uma versão fortemente desfocada da imagem para compor
    o fundo da miniatura (efeito Tilt-Shift)."""

    # pegando informações da imagem: dimensões e cópia
    h, w = img.shape[:2]
    saida = img.copy()

    # processo de redução
    for _ in range(passadas):
        # reduzindo a imagem atual pelo fator
        h_novo = max(1, saida.shape[0] // fator)
        w_novo = max(1, saida.shape[1] // fator)
        reduzida = cv2.resize(saida, (w_novo, h_novo), interpolation=cv2.INTER_AREA)

        # ampliando de volta para o tamanho original com interpolação suave (bicúbica)
        saida = cv2.resize(reduzida, (w, h), interpolation=cv2.INTER_CUBIC)

    return saida

def saturar(img, k=1.6):
    """Aumenta a saturação da imagem multiplicando a distância
    entre cada pixel e o seu valor de cinza correspondente."""

    # convertendo a imagem para float32 para evitar estouro de valores (wrap-around)
    img_f = img.astype(np.float32)

    # obtendo a luminância/tons de cinza de cada pixel (formato 2D)
    cinza = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    # expandindo o canal de cinza para 3 dimensões (h, w, 1) usando broadcasting para casar com BGR
    cinza_3d = cinza[:, :, np.newaxis]

    # afastando cada cor do seu tom de cinza correspondente pelo fator k
      ## Fórmula: cinza + k * (pixel - cinza)
    img_saturada = cinza_3d + k * (img_f - cinza_3d)

    # garantindo que os valores fiquem restritos ao intervalo válido [0, 255] e converter para uint8
    img_saturada = np.clip(img_saturada, 0, 255).astype(np.uint8)

    return img_saturada

def miniatura(img, centro=0.6, largura=0.15, transicao=0.25, fator=8):
    """Gera o efeito miniatura (Tilt-Shift) combinando foco nítido na faixa central,
    desfoque nas bordas e aumento de saturação e contraste."""

    # gerarando a imagem desfocada para o fundo
    fundo = desfoque_reamostragem(img, fator=fator, passadas=2)

    # gerarndo a máscara de foco e convertê-la para o intervalo [0.0, 1.0]
    mascara_uint8 = faixa_de_foco(img, centro=centro, largura=largura, transicao=transicao)
    alfa = mascara_uint8.astype(np.float32) / 255.0

    # expandindo o alfa para 3 canais (h, w, 3) usando broadcasting para aplicar no BGR
    alfa_3d = alfa[:, :, np.newaxis]

    # frente nítida (com cores intensificadas para dar o aspecto de maquete)
      ## aplicamos a saturação
    frente = saturar(img, k=1.6)

    # composição alfa: alfa * frente + (1 - alfa) * fundo
    composicao = alfa_3d * frente.astype(np.float32) + (1.0 - alfa_3d) * fundo.astype(np.float32)

    # convertendo o resultado final de volta para uint8
    saida = np.clip(composicao, 0, 255).astype(np.uint8)

    return saida

def caleidoscopio_8(img, lado=600):
    """Gera um padrão de caleidoscópio com 8 setores simétricos,
    dobrando o quadrante base em sua diagonal principal."""

    # garantindo que a imagem seja quadrada (usando a função)
    img_quadrada = recorte_quadrado(img, lado)

    # definindo o ponto médio
    meio = lado // 2
    q = img_quadrada[:meio, :meio]

    # dobrando o quadrante na diagonal principal
      ## criando coordenadas de grade para o quadrante
    yy, xx = np.mgrid[:meio, :meio]

    # transpondo o quadrante (trocar linhas por colunas preservando os canais BGR)
      ## Em arrays 3D (H, W, C), transpomos os dois primeiros eixos: (1, 0, 2)
    q_transposto = q.transpose(1, 0, 2)

    # Máscara: onde a coluna (xx) for maior que a linha (yy), estamos acima da diagonal
    acima_diagonal = xx > yy

    # criando o setor triangular base combinando o original e o transposto
    setor_base = q.copy()

    # aplicando a máscara 2D diretamente (o NumPy aplica automaticamente nos 3 canais)
    setor_base[acima_diagonal] = q_transposto[acima_diagonal]

    # partindo do setor base dobrado, gerar a simetria de 4 quadrantes
    q1 = setor_base                # Superior Esquerdo
    q2 = setor_base[:, ::-1]       # Superior Direito
    q3 = setor_base[::-1, :]       # Inferior Esquerdo
    q4 = setor_base[::-1, ::-1]    # Inferior Direito

    # juntando os blocos horizontalmente e verticalmente para formar o caleidoscópio de 8
    topo = np.hstack([q1, q2])
    base = np.hstack([q3, q4])
    caleidoscopio = np.vstack([topo, base])

def caleidoscopio_radial(img, setores=6, lado=600):
    """Gera um caleidoscópio radial circular, dividindo a imagem em fatias
    simétricas baseadas em coordenadas polares."""

    # garantindo que a imagem seja quadrada e obtendo centro
    img_q = recorte_quadrado(img, lado)
    centro = lado / 2.0

    # criando grade de coordenadas para todos os pixels da imagem final
    y, x = np.mgrid[:lado, :lado]

    # estabelecendo distâncias horizontal e vertical até o centro
    dx = x - centro
    dy = y - centro

    ## calculando distância e ângulo
    distancia = np.sqrt(dx**2 + dy**2)
    angulo = np.degrees(np.arctan2(dy, dx)) % 360

    ## ajustando o ângulo para a primeira fatia com espelhamento interno
    angulo_fatia = 360.0 / setores
    metade_fatia = angulo_fatia / 2.0

    ### posição do ângulo dentro de uma fatia (0 até angulo_fatia)
    ang_na_fatia = angulo % angulo_fatia

    ### espelhando a segunda metade de cada fatia para garantir a simetria entre vizinhas
    ang_mapeado = np.where(ang_na_fatia > metade_fatia, angulo_fatia - ang_na_fatia, ang_na_fatia)

    ## convertendo de polar (distância e novo ângulo) de volta para cartesiano (linhas e colunas)
    radianos = np.radians(ang_mapeado)
    colunas_origem = centro + distancia * np.cos(radianos)
    linhas_origem = centro + distancia * np.sin(radianos)

    # arredondando para inteiros e restringir aos limites da imagem
    colunas_idx = np.clip(np.round(colunas_origem), 0, lado - 1).astype(np.intp)
    linhas_idx = np.clip(np.round(linhas_origem), 0, lado - 1).astype(np.intp)

    # copiando a cor dos pixels correspondentes da foto de forma vetorizada
    saida = img_q[linhas_idx, colunas_idx]

    # máscara circular para escurecer bordas fora do caleidoscópio
    raio_max = lado / 2.0
    fora_circulo = distancia > raio_max
    saida[fora_circulo] = (0, 0, 0)

    return saida
